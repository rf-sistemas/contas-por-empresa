2) Logo ACIMA da função "async function recebida(m) {", colar esta função:

async function subirGrande(buf, mime, m) {
  const nome = m.message?.documentMessage?.fileName || '';
  const ext = (nome.split('.').pop() || (mime.split('/')[1] || 'bin')).toLowerCase().replace(/[^a-z0-9]/g, '').slice(0, 5);
  const u = await ponte({ acao: 'upload_url', bytes: buf.length, ext, mime });
  if (!u || !u.url) { console.error('upload_url:', u && u.erro); return null; }
  const r = await fetch(u.url, { method: 'PUT', headers: { 'content-type': mime || 'application/octet-stream', 'x-upsert': 'false' }, body: buf });
  if (!r.ok) { console.error('upload grande:', r.status, (await r.text()).slice(0, 200)); return null; }
  return { mime, path: u.path, bytes: buf.length, nome };
}
